function scopeSelect(){return `<label>学習する範囲 <select id="scope"><option value="all" ${scope==='all'?'selected':''}>全30用語</option><option value="chapter" ${scope==='chapter'?'selected':''}>いまの章（第${chapterOf(state.current)+1}章）</option><option value="review" ${scope==='review'?'selected':''}>苦手な用語だけ</option><option value="due" ${scope==='due'?'selected':''}>今日の復習</option></select></label>`;}
function startDeck(){deck=scopedTerms().map(t=>t.id);cursor=0;revealed=false;}
function emptyScope(){return `<div class="surface empty"><h2>対象の用語はありません</h2><p>${scope==='review'?'「まだあやしい」とした用語や、テストで間違えた用語がここに入ります。':scope==='due'?'今日が期限の復習は終わっています。未学習の用語は「全30用語」から学べます。':'別の範囲を選んでください。'}</p><button class="primary" data-all-scope>全30用語から学ぶ</button></div>`;}
function renderFlash(){
 const toolbar=`<div class="toolbar">${scopeSelect()}<button class="tiny" id="shuffle" ${deck.length?'':'disabled'}>順番をシャッフル</button></div>`;
 if(!deck.length){$('content').innerHTML=toolbar+emptyScope();return;}
 const t=TERMS[deck[cursor]-1];
 $('content').innerHTML=toolbar+`<article class="surface"><div class="term-meta"><span class="eyebrow">RECALL · 見ずに説明する</span><span class="quiz-count">${cursor+1} / ${deck.length} 枚</span></div><div class="flash"><span class="pill">第${chapterOf(t.id)+1}章 · ${CHAPTERS[chapterOf(t.id)].short}</span><h2 class="flash-word">${t.name}</h2><p class="term-english">${t.en}</p>${revealed?`<div class="flash-answer">${t.def}</div><p>${t.memory}</p><div class="trap"><strong>混同しやすい点</strong>${t.trap}</div><button class="quiet tiny" data-hide>もう一度、答えを隠す</button>`:'<p class="flash-cue">意味と具体例を、自分の言葉で説明してみよう。</p><button class="primary" data-reveal>答えをめくる</button>'}</div>${revealed?ratingHTML(t):'<p class="notice" style="text-align:center">覚えたかどうかは、答えを見てから記録できます。</p>'}<div class="row" style="justify-content:space-between;margin-top:20px"><button data-flash-prev ${cursor===0?'disabled':''}>← 前</button><button class="quiet tiny" data-term="${t.id}">解説を読む</button><button data-flash-next ${cursor===deck.length-1?'disabled':''}>次 →</button></div>${cursor===deck.length-1&&revealed?'<p class="notice" style="text-align:center;margin-top:18px">この範囲の最後のカードです。理解度を記録したら、確認テストにも挑戦しましょう。</p>':''}</article>`;
}
function startQuiz(){
 const terms=scopedTerms();if(!terms.length)return;
 quiz={items:shuffle(terms.map(t=>t.id)),options:{},index:0,answers:{},finished:false};
 for(const t of terms)quiz.options[t.id]=shuffle(t.choices.map((_,i)=>i));renderQuiz();renderChrome();
}
function renderQuiz(){
 if(!quiz){const count=scopedTerms().length;$('content').innerHTML=`<div class="toolbar">${scopeSelect()}</div>`+(count?`<div class="surface"><span class="eyebrow">PRACTICE · 理解を確かめる</span><h2 style="margin-top:13px">知識を、使ってみよう。</h2><p class="muted">選択式の確認問題を${count}問出題します。各問のあとに、答えの理由を確認できます。</p><div class="connection">全問、この教材用のオリジナル問題です。過去問そのものの難易度・形式を再現する模試ではありません。</div><ul class="points"><li>問題と選択肢は、毎回ランダムな順番です。</li><li>不正解の用語は、自動で「要復習」に入ります。</li><li>正解後は「自分で説明できるか」も考えて記録しましょう。</li></ul><button class="primary" id="quiz-start">${count}問のテストを始める →</button></div>`:emptyScope());return;}
 if(quiz.finished){renderResults();return;}
 const t=TERMS[quiz.items[quiz.index]-1];const a=quiz.answers[t.id];const answered=typeof a==='number';const correct=a===t.correct;
 $('content').innerHTML=`<article class="surface"><div class="term-meta"><span class="eyebrow">QUESTION ${num(quiz.index+1)}</span><span class="quiz-count">${quiz.index+1} / ${quiz.items.length} 問</span></div><div class="progress-track"><div style="width:${quiz.index/quiz.items.length*100}%"></div></div><p class="quiz-question">${t.q}</p><div>${quiz.options[t.id].map((o,i)=>`<button class="option ${answered?(o===t.correct?'correct':o===a?'wrong':''):''}" data-option="${o}" ${answered?'disabled':''}><span class="letter">${['A','B','C'][i]}</span><span>${t.choices[o]}${answered&&o===t.correct?' <strong>✓ 正解</strong>':''}${answered&&o===a&&o!==t.correct?' <strong>× 選んだ答え</strong>':''}</span></button>`).join('')}</div>${answered?`<section class="feedback ${correct?'':'wrong'}" role="status"><strong>${correct?'正解です。':'ここは復習しておこう。'}</strong><p style="margin:7px 0">${t.why}</p><span class="tiny">${num(t.id)} ${t.name}${correct?'':' · 要復習に記録しました'}</span></section>${correct?`<div class="row"><button class="tiny" data-quiz-rate="review" data-id="${t.id}">偶然かも → 要復習にする</button><button class="tiny" data-quiz-rate="known" data-id="${t.id}">理由も説明できる → 記録</button></div>`:''}<div class="row" style="margin-top:24px"><button class="primary" id="quiz-next">${quiz.index===quiz.items.length-1?'結果を見る':'次の問題へ →'}</button></div>`:'<p class="notice" style="margin-top:20px">当てはまるものを一つ選んでください。</p>'}</article><button class="quiet tiny" id="quiz-stop" style="margin-top:13px">テストを終了して範囲を選び直す</button>`;
}
function renderResults(){
 const right=quiz.items.filter(id=>quiz.answers[id]===TERMS[id-1].correct);const wrong=quiz.items.filter(id=>!right.includes(id));
 $('content').innerHTML=`<section class="surface"><span class="eyebrow">RESULT · おつかれさまでした</span><h2 style="margin-top:12px">${wrong.length?'苦手がわかれば、一歩前進。':'この範囲は、全問正解！'}</h2><div class="big-score">${right.length}<span style="font-size:19px" class="muted"> / ${quiz.items.length} 問</span></div><p class="muted">このテストの正答数です。試験本番の得点や合格可能性を表すものではありません。</p>${wrong.length?`<h3>解説を読み直す用語</h3><div class="catalog">${wrong.map(id=>`<button class="catalog-card" data-term="${id}"><span class="mono num">${num(id)}</span><span>${TERMS[id-1].name}</span><span style="margin-left:auto">→</span></button>`).join('')}</div>`:'<p>次は暗記カードで、選択肢なしでも説明できるか試してみよう。</p>'}<div class="row" style="margin-top:23px">${wrong.length?'<button class="primary" id="quiz-retry-wrong">今回間違えた問題だけ再挑戦</button>':''}<button id="quiz-new">範囲を選んでもう一度</button><button id="results-flash">暗記カードで復習</button></div></section>`;
}
function renderList(){

}
function renderCatalog(){
 const text=search.trim().toLocaleLowerCase();const matches=TERMS.filter(t=>(listFilter==='all'||status(t.id)===listFilter)&&(!text||[t.name,t.en,t.read,t.def,t.example,t.memory,...t.points].join(' ').toLocaleLowerCase().includes(text)));
 $('catalog-results').innerHTML=`<p class="tiny muted" role="status">${matches.length} / 30 用語</p>`+(matches.length?CHAPTERS.map((c,i)=>{const ts=matches.filter(t=>chapterOf(t.id)===i);return ts.length?`<section style="margin-bottom:24px"><h3><span class="eyebrow">STEP ${num(i+1)}</span>　${c.name}</h3><div class="catalog">${ts.map(t=>`<button class="catalog-card" data-term="${t.id}"><span class="mono num">${num(t.id)}</span><span class="catalog-name">${t.name}<span class="catalog-desc">${t.memory}</span></span><span class="pill ${status(t.id)==='review'?'review':''}" style="margin-left:auto">${statusName(t.id)}</span></button>`).join('')}</div></section>`:'';}).join(''):'<div class="surface empty"><p>一致する用語はありません。検索語や表示条件を変えてみてください。</p></div>');
}
